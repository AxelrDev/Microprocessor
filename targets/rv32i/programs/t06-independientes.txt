# t06-independientes.txt - no dependency between neighbouring instructions
# and no jumps: the pipeline needs no forwarding, stall or flush.
# Expected: cycles = instructions + 3 (pipeline fill).
addi  x1, x0, 1           # 0x00  x1 = 1
addi  x2, x0, 2           # 0x04  x2 = 2
addi  x3, x0, 3           # 0x08  x3 = 3
lui   x4, 0x12345         # 0x0c  x4 = 0x12345000
auipc x5, 0               # 0x10  x5 = 0x10
addi  x6, x0, -1          # 0x14  x6 = 0xffffffff
xori  x7, x0, 0x55        # 0x18  x7 = 0x55
ori   x8, x0, 0x0f0       # 0x1c  x8 = 0xf0
andi  x9, x6, 0x7ff       # 0x20  x6 written 3 instructions earlier: x9 = 0x7ff
add   x10, x1, x2         # 0x24  x10 = 3
sub   x11, x3, x1         # 0x28  x11 = 2
slli  x12, x2, 4          # 0x2c  x12 = 0x20
ebreak                    # 0x30
